const { Client, GatewayIntentBits } = require('discord.js');

const client = new Client({
    intents: [
        GatewayIntentBits.Guilds,
        GatewayIntentBits.GuildMessages,
        GatewayIntentBits.MessageContent 
    ]
});

// Configure your filters: "CHANNEL_ID": ["word1", "word2"]
const blockedWordsByChannel = {
    "111122223333444455": ["apple", "banana", "orange"], 
    "666677778888999900": ["cats", "dogs", "birds"]       
};

client.on('messageCreate', async (message) => {
    if (message.author.bot || !message.content) return;

    const channelId = message.channel.id;

    if (blockedWordsByChannel[channelId]) {
        const targetWords = blockedWordsByChannel[channelId];
        const textToCheck = message.content.toLowerCase();
        const hasBlockedWord = targetWords.some(word => textToCheck.includes(word));

        if (hasBlockedWord) {
            try {
                await message.delete();
                const warning = await message.channel.send(
                    `${message.author}, your message was deleted because it contained a blocked word.`
                );
                setTimeout(() => warning.delete().catch(() => {}), 5000);
            } catch (error) {
                console.error("Deletion failed:", error);
            }
        }
    }
});

client.once('ready', () => {
    console.log("Your moderation bot is online and working!");
});

client.login(process.env.DISCORD_TOKEN);
